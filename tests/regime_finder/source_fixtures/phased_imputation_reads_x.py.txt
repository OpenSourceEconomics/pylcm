from pylcm import AgeGrid, AgeRange, Regime, Model, DiscreteGrid, Phased, categorical, ScalarInt, fixed_transition

@categorical(ordered=False)
class Binary:
    zero: ScalarInt
    one: ScalarInt

def imputed(x):
    return x

def current_utility(belief):
    return belief

def keep_x(x):
    return x

def dead_utility():
    return 0

def create_model():
    ages = AgeGrid(start=0, inclusive_stop=2, step="Y")
    alive = Regime(
        states={"x": DiscreteGrid(Binary), "belief": Phased(solve=imputed, simulate=DiscreteGrid(Binary))},
        actions={},
        state_transitions={"x": keep_x, "belief": fixed_transition("belief")},
        functions={"utility": current_utility},
    )
    dead = Regime(functions={"utility": dead_utility})
    return Model(regimes={"alive": alive, "dead": dead}, ages=ages,
        initial_nodes=((0, "alive"),),
        edges={"alive": {"alive": AgeRange(exclusive_stop=1), "dead": 1}})
